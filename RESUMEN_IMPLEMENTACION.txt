🗄️  BASE DE DATOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Se crean 2 tablas en MySQL:

TABLA: ordenes
├─ idorden (INT, PK, AUTO_INCREMENT)
├─ idusuario (INT)
├─ fecha_orden (TIMESTAMP)
├─ total (DECIMAL)
├─ estado (VARCHAR) - Estados posibles:
│  ├─ pendiente_pago_tarjeta
│  ├─ pendiente_pago_tienda
│  ├─ pagado
│  └─ cancelado
├─ metodo_pago (VARCHAR) - "tarjeta" o "tienda"
├─ numero_tarjeta (VARCHAR) - Solo últimos 4 dígitos (seguridad)
├─ nombre_tarjeta (VARCHAR)
├─ fecha_vencimiento (VARCHAR)
├─ codigo_barras (VARCHAR UNIQUE) - OED + timestamp + random
├─ tienda_cercana (VARCHAR)
├─ detalles_orden (JSON) - Productos y cantidades
└─ fecha_actualizacion (TIMESTAMP)

TABLA: detalles_orden
├─ iddetalle (INT, PK, AUTO_INCREMENT)
├─ idorden (INT, FK) → ordenes.idorden
├─ idproducto (INT)
├─ nombre_producto (VARCHAR)
├─ precio (DECIMAL)
├─ cantidad (INT)
└─ subtotal (DECIMAL)

🚀 CÓMO EMPEZAR (3 PASOS)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PASO 1: Crear las tablas en la BD
───────────────────────────────────
Opción A (Recomendada - Automática):
  → Abre en el navegador: http://localhost/tiendaonline/crear_tabla_ordenes.php
  → Deberías ver mensajes verdes indicando éxito

Opción B (Manual):
  → Copia el contenido de schema_ordenes.sql
  → Pégalo en phpMyAdmin > SQL
  → Haz clic en Ejecutar

PASO 2: Probar el flujo de compra
──────────────────────────────────
  1. Ve a: http://localhost/tiendaonline/index.php
  2. Agrega productos al carrito
  3. Haz clic en "Mi Carrito"
  4. Haz clic en "Finalizar Compra"
  5. Elige método de pago:
     • Tarjeta: Completa formulario
     • Tienda: Selecciona sucursal
  6. Procesa la compra

PASO 3: Gestionar órdenes (Admin)
──────────────────────────────────
  → Abre: http://localhost/tiendaonline/gestor_ordenes.php
  → Verás todas las órdenes, estadísticas y puedes:
     • Ver detalles de cada orden
     • Ver códigos de barras (para tienda)
     • Marcar como pagada

🎨 CÓDIGO DE BARRAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Formato: CODE128 (Estándar retail internacional)
Patrón:  OED + [timestamp] + [número aleatorio 1000-9999]
Ejemplo: OED1685632801234
